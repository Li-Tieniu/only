## 详情页

### url:/gametrading/details

### 返回

```json
{
itemId: [Long],
wallImages: [String[]],
gameName: [String],
price: [Int],
details: [String]
}
```

## 建库语句

```sql
create table game_trading
(
    item_id      bigint unsigned auto_increment comment '主键ID'
        primary key,
    is_deleted   tinyint unsigned default 0 not null comment '是否删除：0-未删除，1-已删除',
    cover_image varchar(2000)                not null comment '列表展示小图',
    wall_images varchar(2000)                 not null comment '详情展示大图，多图使用$分隔',
    game_name    varchar(50)                  not null comment '游戏名称',
    price        decimal(10, 2)               not null comment '商品价格',
    details      varchar(2000)                not null comment '商品详情',
    create_time  int unsigned                 not null comment '创建时间（时间戳）',
    update_time  int unsigned                 not null comment '修改时间（时间戳）'
)
    comment '游戏交易商品表' collate = utf8mb4_unicode_ci;

create index idx_game_name
    on game_trading (game_name);
```
